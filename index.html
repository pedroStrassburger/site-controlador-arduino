/* ============================================================ CONFIG MQTT — TÓPICOS SEPARADOS ============================================================ */ const BROKER = "broker.hivemq.com"; const PORT = 8884; const CLIENT_ID = "web_" + Math.random().toString(16).substring(2, 10) + "_" + Date.now(); let client = null; // Base = valor do input. Depois viram /cmd e /status let topicoBase = document.getElementById("topicoInput").value.trim(); let topicoCmd = topicoBase + "/cmd"; let topicoStatus = topicoBase + "/status"; /* Elementos (iguais) */ const statusPill = document.getElementById("statusPill"); const statusDot = document.getElementById("statusDot"); const statusText = document.getElementById("statusText"); const led = document.getElementById("led"); const ledIcon = document.getElementById("ledIcon"); const ledLabel = document.getElementById("ledLabel"); const modoLabel = document.getElementById("modoLabel"); const btnOn = document.getElementById("btnOn"); const btnOff = document.getElementById("btnOff"); const btnBlink = document.getElementById("btnBlink"); const btnFade = document.getElementById("btnFade"); const btnSos = document.getElementById("btnSos"); const topicoInput = document.getElementById("topicoInput"); const velocidade = document.getElementById("velocidade"); const velocidadeValor = document.getElementById("velocidadeValor"); const botoes = [btnOn, btnOff, btnBlink, btnFade, btnSos]; /* ---------- Status de rede ---------- */ function setStatus(tipo, texto) { if (tipo === "sync") { statusPill.style.background = "rgba(254, 240, 138, 0.55)"; statusPill.style.color = "#854d0e"; statusDot.className = "pulse-dot w-2.5 h-2.5 rounded-full bg-amber-400"; statusDot.style.boxShadow = "none"; botoes.forEach(b => b.disabled = true); } else if (tipo === "online") { statusPill.style.background = "rgba(167, 243, 208, 0.65)"; statusPill.style.color = "#065f46"; statusDot.className = "w-2.5 h-2.5 rounded-full bg-emerald-500"; statusDot.style.boxShadow = "0 0 10px rgba(16,185,129,0.9)"; botoes.forEach(b => b.disabled = false); } else if (tipo === "offline") { statusPill.style.background = "rgba(254, 202, 202, 0.65)"; statusPill.style.color = "#991b1b"; statusDot.className = "w-2.5 h-2.5 rounded-full bg-red-500"; statusDot.style.boxShadow = "0 0 10px rgba(239,68,68,0.9)"; botoes.forEach(b => b.disabled = true); } statusText.textContent = texto; } /* ---------- LED virtual (só muda aqui) ---------- */ const classesLED = ["led-off", "led-on", "led-blink", "led-fade", "led-sos"]; function aplicarEstadoLED(valor) { const v = valor.toUpperCase().trim(); led.classList.remove(...classesLED); led.style.animation = ""; if (v === "1" || v === "ON") { led.classList.add("led-on"); ledIcon.style.stroke = "rgba(255,255,255,0.95)"; ledLabel.textContent = "Dispositivo Ativo"; ledLabel.style.color = "#1B4332"; ledLabel.style.fontWeight = "600"; } else if (v === "0" || v === "OFF") { led.classList.add("led-off"); ledIcon.style.stroke = "rgba(255,255,255,0.85)"; ledLabel.textContent = "Dispositivo Inativo"; ledLabel.style.color = "#3A4A42"; ledLabel.style.fontWeight = "500"; } else if (v.startsWith("BLINK")) { const ms = v.includes("_") ? v.split("_")[1] : velocidade.value; led.style.setProperty("--blink-speed", ms + "ms"); led.classList.add("led-blink"); ledIcon.style.stroke = "rgba(255,255,255,0.95)"; ledLabel.textContent = "Piscando"; ledLabel.style.color = "#0e7490"; ledLabel.style.fontWeight = "600"; } else if (v.startsWith("FADE")) { const ms = v.includes("_") ? v.split("_")[1] : velocidade.value; led.style.setProperty("--blink-speed", ms + "ms"); led.classList.add("led-fade"); ledIcon.style.stroke = "rgba(255,255,255,0.95)"; ledLabel.textContent = "Efeito Respiração"; ledLabel.style.color = "#6d28d9"; ledLabel.style.fontWeight = "600"; } else if (v === "SOS") { led.classList.add("led-sos"); ledIcon.style.stroke = "rgba(255,255,255,0.95)"; ledLabel.textContent = "SOS • Emergência"; ledLabel.style.color = "#991b1b"; ledLabel.style.fontWeight = "600"; } modoLabel.textContent = "modo: " + v; } /* ---------- Conexão ---------- */ function conectar() { setStatus("sync", "Sincronizando..."); client = new Paho.MQTT.Client(BROKER, PORT, CLIENT_ID); client.onConnectionLost = (resp) => { setStatus("offline", "Sistema Offline"); console.warn("Conexão perdida:", resp.errorMessage); setTimeout(conectar, 4000); }; client.onMessageArrived = (msg) => { const valor = msg.payloadString.trim(); const topico = msg.destinationName; console.log(`📩 Recebido [${topico}]: ${valor}`); /* 🔒 Só aplica se vier do /status */ if (topico === topicoStatus) { aplicarEstadoLED(valor); } }; client.connect({ useSSL: true, onSuccess: () => { setStatus("online", "Sistema Online"); client.subscribe(topicoStatus); // ⬅️ só escuta /status console.log("✅ Conectado. Escutando:", topicoStatus); }, onFailure: (err) => { setStatus("offline", "Falha na Conexão"); console.error("Erro MQTT:", err.errorMessage); setTimeout(conectar, 4000); }, }); } /* ---------- Troca dinâmica de tópico ---------- */ topicoInput.addEventListener("change", () => { const novo = topicoInput.value.trim(); if (!novo || novo === topicoBase) return; if (client && client.isConnected()) { client.unsubscribe(topicoStatus); topicoBase = novo; topicoCmd = topicoBase + "/cmd"; topicoStatus = topicoBase + "/status"; client.subscribe(topicoStatus); console.log(`🔄 Novo status: ${topicoStatus}`); } else { topicoBase = novo; topicoCmd = topicoBase + "/cmd"; topicoStatus = topicoBase + "/status"; } }); /* ---------- Slider ---------- */ velocidade.addEventListener("input", () => { velocidadeValor.textContent = velocidade.value + " ms"; led.style.setProperty("--blink-speed", velocidade.value + "ms"); }); /* ---------- Publicar — sempre no /cmd ---------- */ function publicar(valor) { if (!client || !client.isConnected()) return; const msg = new Paho.MQTT.Message(valor); msg.destinationName = topicoCmd; // ⬅️ /cmd client.send(msg); console.log(`📤 Enviado [${topicoCmd}]: ${valor}`); } btnOn.addEventListener("click", () => publicar("1")); btnOff.addEventListener("click", () => publicar("0")); btnBlink.addEventListener("click", () => publicar("BLINK_" + velocidade.value)); btnFade.addEventListener("click", () => publicar("FADE_" + velocidade.value)); btnSos.addEventListener("click", () => publicar("SOS")); window.addEventListener("load", conectar);
