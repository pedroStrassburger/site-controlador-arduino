import serial import serial.tools.list_ports import paho.mqtt.client as mqtt import time # Configurações de Conexão Serial # DICA: Altere 'COM3' para a porta onde o seu Arduino está conectado (ex: 'COM4' no Windows ou '/dev/ttyUSB0' no Linux) SERIAL_PORT = 'COM3' BAUD_RATE = 115200 # Configurações do Broker HiveMQ Público MQTT_BROKER = "broker.hivemq.com" MQTT_PORT = 1883 TOPIC_POT = "projeto_arduino/potenciometro" TOPIC_LED = "projeto_arduino/led" def setup_serial(): try: ser = serial.Serial(SERIAL_PORT, BAUD_RATE, timeout=0.1) print(f"[SERIAL] Conectado com sucesso na porta {SERIAL_PORT}") return ser except Exception as e: print(f"[SERIAL ERROR] Não foi possível abrir a porta {SERIAL_PORT}: {e}") print("[SERIAL INFO] Portas disponíveis:") for port in serial.tools.list_ports.comports(): print(f" - {port.device}") return None # Callbacks do MQTT def on_connect(client, userdata, flags, rc): if rc == 0: print("[MQTT] Conectado com sucesso ao HiveMQ Broker!") client.subscribe(TOPIC_LED) print(f"[MQTT] Inscrito no tópico de controle: {TOPIC_LED}") else: print(f"[MQTT ERROR] Falha na conexão. Código: {rc}") def on_message(client, userdata, msg): try: payload = msg.payload.decode('utf-8') print(f"[MQTT RX] Tópico: {msg.topic} -> Valor recebido: {payload}") if ser and ser.is_open: # Envia o comando formatado para o Arduino command = f"LED:{payload}\n" ser.write(command.encode('utf-8')) except Exception as e: print(f"[ERROR] Falha ao processar mensagem MQTT: {e}") # Inicialização do Cliente MQTT client = mqtt.Client(mqtt.CallbackAPIVersion.VERSION1, client_id="Python_Arduino_Bridge") client.on_connect = on_connect client.on_message = on_message print("[SYSTEM] Iniciando ponte Serial <-> MQTT...") ser = setup_serial() try: client.connect(MQTT_BROKER, MQTT_PORT, 60) client.loop_start() while True: if ser and ser.is_open and ser.in_waiting > 0: try: line = ser.readline().decode('utf-8', errors='ignore').strip() if line.startswith("POT:"): pot_value = line.split(":")[1] print(f"[SERIAL RX] Valor Potenciômetro: {pot_value}") client.publish(TOPIC_POT, pot_value) except Exception as e: print(f"[SERIAL READ ERROR] {e}") time.sleep(0.01) except KeyboardInterrupt: print("\n[SYSTEM] Encerrando aplicação...") if ser: ser.close() client.loop_stop() client.disconnect() print("[SYSTEM] Desconectado.")
